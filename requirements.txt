# Runtime dependencies.
#
#   pip install -r requirements.txt
#
# PyObjC ships as one coordinated release train, so every package below is held
# at the same version. Pinning only the four the app imports would let pip
# resolve the other four to a newer release, producing an install that mixes two
# PyObjC versions.
#
# Held at 12.2.1 because every measurement recorded in docs/ was taken
# on it. The 12.2.2 release changes only how PyObjC itself is built (Xcode 27
# support) and carries no runtime change, so there is nothing to gain by moving.

# --- imported directly by the app ---
pyobjc-framework-AVFoundation==12.2.1   # AVAudioEngine, AVCaptureDevice
pyobjc-framework-Cocoa==12.2.1          # Foundation (NSRunLoop, NSDate) and AppKit (NSStatusBar)
pyobjc-framework-Speech==12.2.1         # SFSpeechRecognizer, SFSpeechAudioBufferRecognitionRequest
pyobjc-framework-libdispatch==12.2.1    # dispatch_async, dispatch_get_main_queue

# --- pulled in by the four above, pinned to keep the set on one release ---
pyobjc-core==12.2.1                     # the compiled bridge under every binding
pyobjc-framework-CoreAudio==12.2.1
pyobjc-framework-CoreMedia==12.2.1
pyobjc-framework-Quartz==12.2.1
